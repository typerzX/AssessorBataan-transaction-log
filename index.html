/** * Apps Script backend for the Transaction Receiving Log. * HEADERS order (fixed): * Control No. | Date | Received | Office | Event/Subject | Date of Event | Officer in Charge | Classification | Remarks * * This version reads the sheet's header row and maps each data row to the * fixed HEADERS order, so data always appears in the correct columns * regardless of sheet column arrangement. */ const HEADERS = [ "Control No.", "Date", "Received", "Office", "Event/Subject", "Date of Event", "Officer in Charge", "Classification", "Remarks" ]; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheetName = data.sheetName; // e.g. "January" const sheet = getOrCreateSheet(sheetName); // Append in exactly the HEADERS order (ensures consistency) sheet.appendRow([ data.controlNo || "", data.date || "", data.received || "", data.office || "", data.eventSubject || "", data.dateOfEvent || "", data.officerInCharge || "", data.classification || "", data.remarks || "" ]); return ContentService .createTextOutput(JSON.stringify({ status: "success", sheet: sheetName })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const requestedSheet = e.parameter.sheet; const sheetName = requestedSheet || currentMonthName(); const sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService .createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } // Get all values (including header row) const allValues = sheet.getDataRange().getValues(); if (allValues.length < 2) { // Only header or empty – return empty array return ContentService .createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } const headerRow = allValues[0]; const dataRows = allValues.slice(1); // Build a map: header name -> column index (0‑based) const headerIndexMap = {}; headerRow.forEach((col, idx) => { // Trim to handle any accidental spaces const key = String(col).trim(); if (key) headerIndexMap[key] = idx; }); // Now for each data row, extract values in the order of HEADERS const formattedRows = dataRows.map(row => { // Build an array of 9 values in HEADERS order const mappedRow = HEADERS.map(header => { const idx = headerIndexMap[header]; if (idx === undefined) return ""; // header not found -> empty let cell = row[idx]; // Format Date objects to YYYY-MM-DD if (cell instanceof Date) { const year = cell.getFullYear(); const month = String(cell.getMonth() + 1).padStart(2, '0'); const day = String(cell.getDate()).padStart(2, '0'); return `${year}-${month}-${day}`; } return cell !== undefined ? cell : ""; }); return mappedRow; }); return ContentService .createTextOutput(JSON.stringify(formattedRows)) .setMimeType(ContentService.MimeType.JSON); } // ---- helpers ---- function getOrCreateSheet(sheetName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight("bold"); } return sheet; } function currentMonthName() { const months = ["January","February","March","April","May","June", "July","August","September","October","November","December"]; return months[new Date().getMonth()]; }
